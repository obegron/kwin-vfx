cmake_minimum_required(VERSION 3.24)
project(kwin_vfx VERSION 0.0.1 LANGUAGES NONE)

option(KWIN_VFX_BUILD_EFFECT "Build the experimental native effect" ON)
option(KWIN_VFX_CHECK_HEADERS "Compile the KWin interface probe (no plugin)" OFF)
option(KWIN_VFX_BUILD_SMOKE_CLIENT "Build the isolated compositor test client" OFF)
option(KWIN_VFX_BUILD_SHADER_TESTS "Build the independent GL shader tests and benchmark" OFF)
option(KWIN_VFX_BUILD_NVIDIA_PROBES "Build standalone CUDA/GL and native VFX probes (no effect integration)" OFF)
option(KWIN_VFX_BUILD_NVIDIA_BACKEND "Build the optional NVIDIA worker backend (explicit desktop opt-in)" OFF)
option(KWIN_VFX_BUILD_PREVIEW_TRAY "Build window enhancement tray controls (desktop or private preview)" OFF)
if(KWIN_VFX_BUILD_NVIDIA_BACKEND AND NOT KWIN_VFX_BUILD_EFFECT)
    message(FATAL_ERROR "The NVIDIA backend requires KWIN_VFX_BUILD_EFFECT=ON")
endif()
if(KWIN_VFX_BUILD_EFFECT OR KWIN_VFX_CHECK_HEADERS OR KWIN_VFX_BUILD_SMOKE_CLIENT OR KWIN_VFX_BUILD_SHADER_TESTS OR KWIN_VFX_BUILD_NVIDIA_PROBES OR KWIN_VFX_BUILD_PREVIEW_TRAY)
    enable_language(CXX)
    include(GNUInstallDirs)
endif()

if(KWIN_VFX_BUILD_PREVIEW_TRAY)
    find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets DBus)
    add_executable(kwin_vfx_preview_tray src/preview/tray.cpp)
    target_compile_features(kwin_vfx_preview_tray PRIVATE cxx_std_20)
    target_compile_options(kwin_vfx_preview_tray PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(kwin_vfx_preview_tray PRIVATE Qt6::Widgets Qt6::DBus)
    install(PROGRAMS $<TARGET_FILE:kwin_vfx_preview_tray> DESTINATION bin RENAME kwin-vfx-tray)
endif()

if(KWIN_VFX_BUILD_NVIDIA_BACKEND)
    find_package(CUDAToolkit REQUIRED)
    find_package(Threads REQUIRED)
endif()

if(KWIN_VFX_BUILD_NVIDIA_PROBES)
    find_package(Qt6 REQUIRED COMPONENTS Core Gui OpenGL)
    find_package(CUDAToolkit REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
    add_executable(kwin_vfx_cuda_gl_probe src/backends/nvidia/interop_probe.cpp)
    target_compile_features(kwin_vfx_cuda_gl_probe PRIVATE cxx_std_20)
    target_compile_options(kwin_vfx_cuda_gl_probe PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(kwin_vfx_cuda_gl_probe PRIVATE Qt6::Gui Qt6::OpenGL CUDA::cudart)
    add_executable(kwin_vfx_native_vfx_probe src/backends/nvidia/native_vfx_probe.cpp)
    target_compile_features(kwin_vfx_native_vfx_probe PRIVATE cxx_std_20)
    target_compile_options(kwin_vfx_native_vfx_probe PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(kwin_vfx_native_vfx_probe PRIVATE Qt6::Gui CUDA::cuda_driver ${CMAKE_DL_LIBS})
    add_executable(kwin_vfx_worker_probe tests/nvidia_worker_probe.cpp src/backends/nvidia/vfx_worker.cpp)
    target_include_directories(kwin_vfx_worker_probe PRIVATE src/backends/nvidia)
    target_compile_features(kwin_vfx_worker_probe PRIVATE cxx_std_20)
    target_compile_options(kwin_vfx_worker_probe PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(kwin_vfx_worker_probe PRIVATE Qt6::Gui Qt6::OpenGL CUDA::cuda_driver PkgConfig::EPOXY ${CMAKE_DL_LIBS})
endif()
if(KWIN_VFX_BUILD_EFFECT OR KWIN_VFX_CHECK_HEADERS OR KWIN_VFX_BUILD_SMOKE_CLIENT)
    find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets Quick DBus)
    find_package(KWin 6.8.1 EXACT REQUIRED)
endif()

if(KWIN_VFX_BUILD_SHADER_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Core Gui OpenGL)
    set(CMAKE_AUTORCC ON)
    add_executable(kwin_vfx_shader_probe tests/shader_probe.cpp src/backends/shader/filter.qrc)
    target_compile_features(kwin_vfx_shader_probe PRIVATE cxx_std_20)
    target_compile_options(kwin_vfx_shader_probe PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(kwin_vfx_shader_probe PRIVATE Qt6::Gui Qt6::OpenGL)
    add_executable(kwin_vfx_profile_reset_test tests/profile_reset.cpp src/effect/renderprofile.cpp)
    target_include_directories(kwin_vfx_profile_reset_test PRIVATE src/effect)
    target_compile_features(kwin_vfx_profile_reset_test PRIVATE cxx_std_20)
    target_link_libraries(kwin_vfx_profile_reset_test PRIVATE Qt6::Core)
    if(KWIN_VFX_BUILD_EFFECT)
        add_executable(kwin_vfx_hdr_shader_probe tests/hdr_shader_probe.cpp src/effect/shaders.qrc)
        target_compile_features(kwin_vfx_hdr_shader_probe PRIVATE cxx_std_23)
        target_link_libraries(kwin_vfx_hdr_shader_probe PRIVATE Qt6::Gui Qt6::OpenGL KWin::kwin)
    endif()
endif()

if(KWIN_VFX_BUILD_SMOKE_CLIENT)
    add_executable(kwin_vfx_smoke_client tests/smoke_client.cpp)
    target_link_libraries(kwin_vfx_smoke_client PRIVATE Qt6::Widgets)
endif()
if(KWIN_VFX_CHECK_HEADERS)
    add_library(kwin_vfx_header_probe OBJECT tests/kwin_headers.cpp)
    target_compile_features(kwin_vfx_header_probe PRIVATE cxx_std_23)
    target_link_libraries(kwin_vfx_header_probe PRIVATE KWin::kwin)
endif()

if(KWIN_VFX_BUILD_EFFECT)
    find_package(KF6GlobalAccel 6.30 REQUIRED)
    set(CMAKE_AUTOMOC ON)
    set(CMAKE_AUTORCC ON)
    add_subdirectory(src/effect)
endif()

if(KWIN_VFX_BUILD_PREVIEW_TRAY)
    if(IS_ABSOLUTE "${KDE_INSTALL_PLUGINDIR}")
        set(KWIN_VFX_PLUGIN_ROOT "${KDE_INSTALL_PLUGINDIR}")
    else()
        set(KWIN_VFX_PLUGIN_ROOT "${CMAKE_INSTALL_PREFIX}/${KDE_INSTALL_PLUGINDIR}")
    endif()
    set(KWIN_VFX_NVIDIA_JSON false)
    set(KWIN_VFX_USER_PLUGIN_PATH_JSON false)
    if(KWIN_VFX_PLUGIN_ROOT MATCHES "/kwin-vfx/plugins$")
        set(KWIN_VFX_USER_PLUGIN_PATH_JSON true)
    endif()
    if(KWIN_VFX_BUILD_NVIDIA_BACKEND)
        set(KWIN_VFX_NVIDIA_JSON true)
    endif()
    configure_file(packaging/installation.json.in installation.json @ONLY)
    configure_file(packaging/kwin-vfx-tray.desktop.in kwin-vfx-tray.desktop @ONLY)
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/installation.json" DESTINATION "${CMAKE_INSTALL_DATADIR}/kwin-vfx")
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/kwin-vfx-tray.desktop" DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
    install(PROGRAMS scripts/setup-desktop.py DESTINATION "${CMAKE_INSTALL_BINDIR}" RENAME kwin-vfx-setup)
    install(PROGRAMS scripts/prepare-nvidia-runtime.py DESTINATION "${CMAKE_INSTALL_BINDIR}" RENAME kwin-vfx-runtime)
    install(FILES LICENSES/GPL-2.0-or-later.txt LICENSES/MIT-AMD-CAS.txt DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/kwin-vfx")
endif()
