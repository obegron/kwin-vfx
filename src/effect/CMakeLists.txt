set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
include(GNUInstallDirs)
if(NOT KDE_INSTALL_PLUGINDIR)
    set(KDE_INSTALL_PLUGINDIR "${CMAKE_INSTALL_LIBDIR}/qt6/plugins" CACHE PATH "Qt plugin installation directory" FORCE)
endif()
kcoreaddons_add_plugin(kwin_vfx
    SOURCES windoweffect.cpp windoweffect.h main.cpp renderprofile.cpp renderprofile.h shaders.qrc ../backends/shader/filter.qrc
    INSTALL_NAMESPACE "kwin/effects/plugins")
target_compile_features(kwin_vfx PRIVATE cxx_std_23)
set_target_properties(kwin_vfx PROPERTIES PREFIX "")
target_compile_options(kwin_vfx PRIVATE -Wall -Wextra -Wpedantic)
find_package(KDecoration3 REQUIRED)
target_link_libraries(kwin_vfx PRIVATE KWin::kwin KF6::GlobalAccel KDecoration3::KDecoration Qt6::DBus)
if(KWIN_VFX_BUILD_NVIDIA_BACKEND)
    target_sources(kwin_vfx PRIVATE nvidiaslot.cpp nvidiaslot.h ../backends/nvidia/vfx_worker.cpp)
    target_compile_definitions(kwin_vfx PRIVATE KWIN_VFX_NVIDIA_BACKEND=1)
    target_link_libraries(kwin_vfx PRIVATE CUDA::cuda_driver Threads::Threads ${CMAKE_DL_LIBS})
endif()
set_property(TARGET kwin_vfx APPEND PROPERTY AUTOMOC_MACRO_NAMES KWIN_EFFECT_FACTORY_SUPPORTED)
